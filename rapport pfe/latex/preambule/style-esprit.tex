\setstretch{1.15}
\setlength{\parindent}{0.5cm}
\setlength{\parskip}{0.15\baselineskip}

% ---------------------------------------------------------------- Couleurs
%  La structure du document est en noir. Les rares couleurs restantes
%  servent aux encadres de contenu et aux emplacements reserves.
\definecolor{noirTitre}{HTML}{000000}
\definecolor{grisFilet}{HTML}{8A8A8A}
\definecolor{grisFond}{HTML}{F2F2F2}
\definecolor{bleuTitre}{HTML}{000000}   % conserve pour compatibilite

% ---------------------------------------------------------------- Chapitres
%  Presentation reprise du rapport de reference :
%     barre noire epaisse portant « Chapitre X »
%     filet fin
%     titre du chapitre, centre, en grandes capitales
%     filet fin
\newlength{\barreChap}
\setlength{\barreChap}{0.36\textwidth}

%  Aucun \par dans ce qui suit : titlesec lit ses arguments en mode
%  court, une fin de paragraphe y provoquerait l'erreur
%  « Paragraph ended before \ttl@format@ii was complete ».
\newcommand{\filetLarge}{%
  \noindent{\color{grisFilet}\rule{\textwidth}{0.4pt}}%
}

\newcommand{\enteteChapitre}[1]{%
  \noindent
  \begin{minipage}{\textwidth}
    \centering
    \rule[0.30\baselineskip]{\barreChap}{3.5pt}%
    \hspace{0.5em}{\normalsize\bfseries #1}\hspace{0.5em}%
    \rule[0.30\baselineskip]{\barreChap}{3.5pt}%
  \end{minipage}%
}

\titleformat{\chapter}[display]
  {\normalfont\bfseries\color{noirTitre}\centering}
  {\enteteChapitre{\MakeUppercase{Chapitre \thechapter}}}
  {12pt}
  {\filetLarge\\[10pt]\color{noirTitre}\LARGE\MakeUppercase}
  [\vspace{8pt}\filetLarge]
\titlespacing*{\chapter}{0pt}{6pt}{34pt}

%  Chapitres non numerotes : meme filet, sans la barre « CHAPITRE X »
\titleformat{name=\chapter,numberless}[display]
  {\normalfont\bfseries\color{noirTitre}\centering}
  {}
  {0pt}
  {\filetLarge\\[10pt]\color{noirTitre}\LARGE\MakeUppercase}
  [\vspace{8pt}\filetLarge]

% ---------------------------------------------------------------- Sections
\titleformat{\section}{\normalfont\large\bfseries\color{noirTitre}}
  {\thesection}{0.6em}{}
\titleformat{\subsection}{\normalfont\normalsize\bfseries\color{noirTitre}}
  {\thesubsection}{0.6em}{}
\titleformat{\subsubsection}{\normalfont\small\bfseries\color{noirTitre}}
  {\thesubsubsection}{0.6em}{}

\titlespacing*{\section}      {0pt}{16pt}{7pt}
\titlespacing*{\subsection}   {0pt}{12pt}{5pt}
\titlespacing*{\subsubsection}{0pt}{10pt}{4pt}

\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{2}

% ---------------------------------------------------------------- En-tetes
%  En-tete : numero et nom du chapitre courant, souligne d'un filet fin.
%  Pied : numero de page et nombre total, centre.
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[L]{\small\nouppercase{\leftmark}}
\fancyfoot[C]{\small\thepage/\pageref{LastPage}}

%  La classe report impose le style « plain » sur la premiere page de
%  chaque chapitre. Sans cette redefinition, ces pages retombent sur le
%  numero seul, centre : c'est ce qui donnait « 18 » au lieu de « 18/85 ».
\fancypagestyle{plain}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
  \fancyfoot[C]{\small\thepage/\pageref{LastPage}}%
}

\renewcommand{\chaptermark}[1]{\markboth{\thechapter.\ #1}{}}

% ---------------------------------------------------------------- Legendes
\captionsetup[figure]{name=Figure}
\captionsetup[table]{name=Tableau}
\renewcommand{\cftfigpresnum}{Figure~}
\renewcommand{\cfttabpresnum}{Tableau~}
\setlength{\cftfignumwidth}{4.2em}
\setlength{\cfttabnumwidth}{4.6em}

\captionsetup[table]{position=above,skip=4pt}
\captionsetup[figure]{position=below,skip=6pt}

% ---------------------------------------------------- Pages liminaires
%  POURQUOI DES PAGES BLANCHES APPARAISSENT
%  La classe est chargee en « twoside, openright » : chaque nouvelle
%  piece commence sur une page de DROITE, comme il est d'usage pour un
%  document imprime en recto-verso. Quand la piece precedente se termine
%  elle-meme sur une page de droite, LaTeX intercale une page blanche.
%  C'est voulu, et c'est ce que demande le guide, qui recommande
%  l'impression recto-verso.
%
%  Les CHAPITRES et les grandes divisions de fin conservent ce
%  comportement. Les pages liminaires, elles, n'ont pas cette exigence :
%      \clearpage        -> pas de page blanche entre elles  (defaut)
%      \cleardoublepage  -> chacune sur une page de droite
\newcommand{\sautLiminaire}{\clearpage}

% ---------------------------------------------------------------- Flottants
%  Reglages d'origine de LaTeX : un flottant en tete de page ne peut
%  occuper que 70 % de la hauteur, et toute page portant un flottant
%  doit conserver 20 % de texte. Deux diagrammes qui se suivent
%  franchissent aussitot ces seuils : ils sont alors reportes sur une
%  page de flottants, qu'ils ne remplissent qu'a moitie. Desserrer ces
%  bornes supprime ces demi-pages vides.
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.75}
\renewcommand{\textfraction}{0.07}
\renewcommand{\floatpagefraction}{0.80}
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{4}
